   131→export function ModeSelector({ mode, onChange, isRecording }: ModeSelectorProps) {
   132→  const modes: { id: CameraMode; label: string; icon: React.ReactNode }[] = [
   133→    { id: "photo", label: "FOTO", icon: <Camera className="size-3.5" /> },
   134→    { id: "portrait", label: "PORTRAIT", icon: <Aperture className="size-3.5" /> },
   135→    { id: "live", label: "LIVE", icon: <CircleDot className="size-3.5" /> },
   136→    { id: "burst", label: "BURST", icon: <Layers className="size-3.5" /> },
   137→    { id: "video", label: "VIDEO", icon: <Video className="size-3.5" /> },
   138→  ];
   139→  return (
   140→    <div className="flex items-center justify-center gap-1 overflow-x-auto no-scrollbar">
   141→      {modes.map((m) => {
   142→        const active = mode === m.id;
   143→        return (
   144→          <button
   145→            key={m.id}
   146→            disabled={isRecording}
   147→            onClick={() => onChange(m.id)}
   148→            className={cn(
   149→              "px-3 py-1.5 rounded-full flex items-center gap-1.5 text-[11px] font-bold tracking-wider transition-all flex-shrink-0",
   150→              active
   151→                ? "bg-amber-300 text-black"
   152→                : "text-white/70 hover:text-white",
   153→              isRecording && !active && "opacity-30",
   154→            )}
   155→          >
   156→            {m.icon}
   157→            {m.label}
   158→          </button>
   159→        );
   160→      })}
   161→    </div>
   162→  );
   163→}

   165→interface CaptureButtonProps {
   166→  mode: CameraMode;
   167→  isRecording: boolean;

   169→  recordingSeconds: number;
   170→  timerCountdown?: number | null;

   173→
   174→export function CaptureButton({
   175→  mode,
   176→  isRecording,
   177→  onCapture,
   178→  recordingSeconds,
   179→  timerCountdown,
   180→  burstCount,
   181→}: CaptureButtonProps) {
   182→  if (mode === "video") {
   183→    return (
   184→      <button
   185→        onClick={onCapture}
   186→        className="relative size-20 rounded-full flex items-center justify-center active:scale-95 transition-transform"
   187→        aria-label={isRecording ? "Stop recording" : "Start recording"}
   188→      >
   189→        <span className="absolute inset-0 rounded-full border-4 border-white" />
   190→        {isRecording ? (
   191→          <span className="size-8 rounded-md bg-red-500 animate-pulse" />
   192→        ) : (
   193→          <span className="size-14 rounded-full bg-red-500" />
   194→        )}
   195→        {isRecording && (
   196→          <span className="absolute -bottom-7 text-xs font-bold text-red-400 tabular-nums">
   197→            {formatTime(recordingSeconds)}
   198→          </span>
   199→        )}
   200→        {typeof timerCountdown === "number" && timerCountdown > 0 && (
   201→          <span className="absolute inset-0 flex items-center justify-center text-3xl font-bold text-amber-300">
   202→            {timerCountdown}
   203→          </span>
   204→        )}
   205→      </button>
   206→    );
   207→  }
   208→
   209→  // Photo / Live / Burst / Portrait
   210→  const ringColor =
   211→    mode === "live"
   212→      ? "border-amber-300"
   213→      : mode === "burst"
   214→        ? "border-sky-300"
   215→        : mode === "portrait"
   216→          ? "border-fuchsia-300"
   217→          : "border-white";
   218→
   219→  const fillColor =
   220→    mode === "live"
   221→      ? "bg-amber-300"
   222→      : mode === "burst"
   223→        ? "bg-sky-300"
   224→        : mode === "portrait"
   225→          ? "bg-fuchsia-300"
   226→          : "bg-white";
   227→
   228→  return (
   229→    <button
   230→      onClick={onCapture}
   231→      className="relative size-20 rounded-full flex items-center justify-center active:scale-95 transition-transform"
   232→      aria-label="Capture"
   233→    >
   234→      <span className={cn("absolute inset-0 rounded-full border-4", ringColor)} />
   235→      <span className={cn("size-16 rounded-full", fillColor)} />
   236→      {typeof timerCountdown === "number" && timerCountdown > 0 && (
   237→        <span className="absolute inset-0 flex items-center justify-center text-4xl font-black text-black animate-ping-slow">
   238→          {timerCountdown}
   239→        </span>
   240→      )}
   241→      {typeof burstCount === "number" && burstCount > 0 && (
   242→        <span className="absolute inset-0 flex items-center justify-center text-2xl font-black text-black">
   243→          {burstCount}
   244→        </span>
   245→      )}
   246→      {mode === "live" && (
   247→        <span className="absolute -bottom-7 text-[10px] font-bold tracking-wider text-amber-300">
   248→          LIVE PHOTO
   249→        </span>
   250→      )}
   251→      {mode === "burst" && (
   252→        <span className="absolute -bottom-7 text-[10px] font-bold tracking-wider text-sky-300">
   253→          BURST (5×)
   254→        </span>
   255→      )}
   256→      {mode === "portrait" && (
   257→        <span className="absolute -bottom-7 text-[10px] font-bold tracking-wider text-fuchsia-300">
   258→          PORTRAIT
   259→        </span>
   260→      )}
   261→    </button>
   262→  );
   263→}
   264→
   265→function formatTime(sec: number): string {
   266→  const m = Math.floor(sec / 60);
   267→  const s = sec % 60;
   268→  return `${m}:${s.toString().padStart(2, "0")}`;
   269→}
